cmake_minimum_required(VERSION 3.25)
cmake_policy(SET CMP0091 NEW)

project(
    LinuxVstBridgeHP0
    VERSION 0.1.0
    DESCRIPTION "HP0 deterministic native Linux VST3 host probe"
    LANGUAGES C CXX
)

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")

option(WF0_BUILD_ONLY "Build only the authorized WF0 Windows probe and fault fixtures" OFF)

if(WF0_BUILD_ONLY)
    if(NOT WIN32 OR NOT MSVC)
        message(FATAL_ERROR "WF0 requires the approved native Windows MSVC build plane")
    endif()
    if(NOT CMAKE_GENERATOR STREQUAL "Visual Studio 17 2022")
        message(FATAL_ERROR "WF0 requires the Visual Studio 17 2022 generator")
    endif()
    if(NOT CMAKE_GENERATOR_PLATFORM STREQUAL "x64")
        message(FATAL_ERROR "WF0 requires generator platform x64")
    endif()
    if(NOT CMAKE_VS_PLATFORM_TOOLSET STREQUAL "v143")
        message(FATAL_ERROR "WF0 requires platform toolset v143")
    endif()
    if(NOT CMAKE_SYSTEM_VERSION STREQUAL "10.0.19041.0")
        message(FATAL_ERROR "WF0 requires Windows SDK 10.0.19041.0")
    endif()
    if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "WF0 requires an x64 compiler")
    endif()

    include(WF0DependencyLock)
    wf0_verify_vst3_sdk("${WF0_VST3_SDK_ROOT}")

    set(CMAKE_CXX_STANDARD 20)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_EXTENSIONS OFF)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
    set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "No configure-time downloads" FORCE)
    set(SMTG_USE_STATIC_CRT ON CACHE BOOL "Use the approved static MSVC runtime" FORCE)
    set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
    set(SMTG_CREATE_BUNDLE_FOR_WINDOWS OFF CACHE BOOL "" FORCE)
    set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
    set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
    set(SMTG_CREATE_MODULE_INFO OFF CACHE BOOL "" FORCE)

    add_subdirectory("${WF0_VST3_SDK_ROOT}" "${CMAKE_BINARY_DIR}/vst3sdk" EXCLUDE_FROM_ALL)
    add_subdirectory(windows-factory-probe)
    add_subdirectory(windows-fixtures/wf0)
    return()
endif()

option(AP2_BUILD_ONLY "Build the offline native proxy and SDK test host" OFF)
if(AP2_BUILD_ONLY)
    include(HP0Vst3SdkLock)
    hp0_verify_vst3_sdk("${VST3_SDK_ROOT}")
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "x86_64")
        message(FATAL_ERROR "AP2 native bundle requires Linux x86-64")
    endif()
    set(CMAKE_CXX_STANDARD 20)
    set(CMAKE_POSITION_INDEPENDENT_CODE ON)
    set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
    set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
    set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
    set(SMTG_CREATE_MODULE_INFO OFF CACHE BOOL "" FORCE)
    set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "" FORCE)
    add_subdirectory("${VST3_SDK_ROOT}" "${CMAKE_BINARY_DIR}/vst3sdk" EXCLUDE_FROM_ALL)
    include(HP0ModernGcc)
    hp0_remove_legacy_stdcxxfs()
    add_subdirectory(native-vst3-proxy)
    return()
endif()

include(HP0ToolchainGuard)
hp0_require_freedesktop_sdk()

include(HP0Vst3SdkLock)
hp0_verify_vst3_sdk("${VST3_SDK_ROOT}")

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "HP0 requires Linux; observed '${CMAKE_SYSTEM_NAME}'")
endif()
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
    message(FATAL_ERROR "HP0 requires x86_64; observed '${CMAKE_SYSTEM_PROCESSOR}'")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "HP0 requires a 64-bit compiler")
endif()
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "15.2.0")
    message(FATAL_ERROR "HP0 requires Freedesktop SDK GCC 15.2.0; observed '${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}'")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "No configure-time downloads" FORCE)

set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)

add_subdirectory("${VST3_SDK_ROOT}" "${CMAKE_BINARY_DIR}/vst3sdk")

include(HP0ModernGcc)
hp0_remove_legacy_stdcxxfs()

if(NOT TARGET validator)
    message(FATAL_ERROR "Pinned VST3 SDK did not expose the official validator target")
endif()

add_subdirectory(native-probe)
hp0_remove_legacy_stdcxxfs()

add_custom_target(hp0-probe DEPENDS LabHostProbe validator)
