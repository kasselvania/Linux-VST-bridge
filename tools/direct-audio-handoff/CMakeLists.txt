cmake_minimum_required(VERSION 3.24)
project(lvb_direct_audio_probe LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The probe requires x86-64 MSVC/MASM")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
add_executable(lvb-futex-probe probe.cpp)
add_library(lvb-direct-wait SHARED shim.cpp)
target_compile_features(lvb-direct-wait PRIVATE cxx_std_20)
target_compile_options(lvb-direct-wait PRIVATE /W4 /WX /permissive- /EHsc /Brepro)
target_link_options(lvb-direct-wait PRIVATE /Brepro /INCREMENTAL:NO)
target_compile_features(lvb-futex-probe PRIVATE cxx_std_20)
target_compile_options(lvb-futex-probe PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/WX;/permissive-;/EHsc;/Brepro>")
target_link_options(lvb-futex-probe PRIVATE /Brepro /INCREMENTAL:NO)
target_compile_definitions(lvb-futex-probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
